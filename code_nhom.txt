using System;
using System.IO;
using System.Text;


namespace Server
{
    /// <summary>
    /// Ghi nhật ký của server ra màn hình console (và ra tệp nếu bật), mỗi dòng kèm mốc thời gian.
    /// Dùng được từ nhiều tác vụ (mỗi client một tác vụ) cùng lúc nhờ khoá lock.
    /// Lớp này chỉ ghi log, không đụng tới TCP hay SQL.
    /// </summary>
    public static class NhatKy
    {
        private static readonly object khoa = new object();
        private static string duongDanTep;

        /// <summary>
        /// Gọi một lần khi server khởi động.
        /// duongDanTep = null: chỉ ghi ra console; có giá trị (ví dụ "server.log"): ghi thêm ra tệp.
        /// </summary>
        public static void KhoiTao(string duongDanTep = null)
        {
        
            Console.OutputEncoding = Encoding.UTF8;
            NhatKy.duongDanTep = duongDanTep;
        }

        /// <summary>Ghi một dòng: [2026-11-15 15:32:05] nội dung</summary>
        public static void Ghi(string noiDung)
        {
            string dong = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] {noiDung}";

            lock (khoa)
            {
                Console.WriteLine(dong);

                if (duongDanTep != null)
                {
                    try
                    {
                        File.AppendAllText(duongDanTep, dong + Environment.NewLine, Encoding.UTF8);
                    }
                    catch (IOException)
                    {
                        // Lỗi ghi tệp không được làm sập server; console vẫn có dòng log
                    }
                    catch (UnauthorizedAccessException)
                    {
                        // Không có quyền ghi vào đường dẫn đã chọn
                    }
                }
            }
        }

        /// <summary>Ghi một dòng lỗi (thêm chữ LỖI ở đầu cho dễ tìm).</summary>
        public static void Loi(string noiDung)
        {
            Ghi("LỖI: " + noiDung);
        }
    }
}

using Server.Security;
using System;
using System.Data.SqlClient;
using System.Collections.Generic;
using System.Text;
using SharedLibrary;

namespace Server.Handlers
{
    public class LoginHandler
    {
        private readonly string _connectionString = "CHUỖI_KẾT_NỐI_SQL_CỦA_NHÓM";

        public LoginResponse Handle(LoginRequest request)
        {
            var response = new LoginResponse();
            string commonErrorMsg = "Sai tên đăng nhập hoặc mật khẩu";

            using (SqlConnection conn = new SqlConnection(_connectionString))
            {
                string query = "SELECT UserId, Username, PasswordHash, PasswordSalt, FullName, Email FROM NguoiDung WHERE Username = @Username";
                SqlCommand cmd = new SqlCommand(query, conn);
                cmd.Parameters.AddWithValue("@Username", request.Username);

                conn.Open();
                using (SqlDataReader reader = cmd.ExecuteReader())
                {
                    if (reader.Read())
                    {
                        string savedHash = reader["PasswordHash"].ToString();
                        string savedSalt = reader["PasswordSalt"].ToString();

                        bool isValid = PasswordHasher.VerifyPassword(request.Password, savedHash, savedSalt);

                        if (isValid)
                        {
                            response.IsSuccess = true;
                            response.UserInfo = new UserDto
                            {
                                UserId = (int)reader["UserId"],
                                Username = reader["Username"].ToString(),
                                FullName = reader["FullName"].ToString(),
                                Email = reader["Email"].ToString()
                            };
                        }
                        else
                        {
                            response.IsSuccess = false;
                            response.ErrorMessage = commonErrorMsg;
                        }
                    }
                    else
                    {
                        response.IsSuccess = false;
                        response.ErrorMessage = commonErrorMsg;
                    }
                }
            }
            return response;
        }
    }
}

using System;
using System.Collections.Generic;
using System.Security.Cryptography;
using System.Text;

namespace Server.Security
{
    public class PasswordHasher
    {
        private const int SaltSize = 16; 
        private const int KeySize = 32;  
        private const int Iterations = 100000; 

        public static void HashPassword(string password, out string hash, out string salt)
        {
            using (var algorithm = new Rfc2898DeriveBytes(password, SaltSize, Iterations, HashAlgorithmName.SHA256))
            {
                salt = Convert.ToBase64String(algorithm.Salt);
                hash = Convert.ToBase64String(algorithm.GetBytes(KeySize));
            }
        }

        public static bool VerifyPassword(string password, string savedHash, string savedSalt)
        {
            byte[] saltBytes = Convert.FromBase64String(savedSalt);
            using (var algorithm = new Rfc2898DeriveBytes(password, saltBytes, Iterations, HashAlgorithmName.SHA256))
            {
                byte[] keyToCheck = algorithm.GetBytes(KeySize);
                byte[] originalKey = Convert.FromBase64String(savedHash);

                return CryptographicOperations.FixedTimeEquals(keyToCheck, originalKey);
            }
        }
    }
}
